// npm install @telenotify/node
import { Telenotify } from '@telenotify/node';

await telenotify.messages.send({
  to: 'customer_123',
  template: 'payment_received',
  params: { amount: '1,250', payer: 'Abebe' },
});

from telenotify import Telenotify

  template='payment_received',
  params={'amount': '1,250', 'payer': 'Abebe'},
)
